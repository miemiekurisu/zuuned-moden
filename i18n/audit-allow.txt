# i18n/audit-allow.txt - the review surface for wording that is NOT interface text.
#
# Every line here says: this literal is not a sentence the user reads, and here is
# why. i18n/audit.py refuses (exit 1) any sentence-shaped literal under src/ or
# transcode/ that is neither inside UiText::tr() nor named here, so this file is the
# whole argument for "language lives in i18n/zh_CN.tsv, not in the code". Keep it
# short and keep every reason true - an unreviewed entry is a hidden UI string.
#
#   SKIP-FILE :: <repo path> :: <reason>
#   SKIP-LINE :: <regex>     :: <reason>   (a matching source line is not wording)
#   <literal>                :: <reason>   (exact, or /regex/)

# --- output that never reaches a widget -------------------------------------
SKIP-LINE :: \bfprintf\s*\( :: diagnostics written to stderr; the session log, not the UI
SKIP-LINE :: \bprintf\s*\( :: C-level stdout/stderr in the transcode and CLI tools
SKIP-LINE :: \b(?:qWarning|qDebug|qInfo|qCritical|qFatal)\s*\( :: Qt log categories, never rendered
SKIP-LINE :: SessionLog|sessionLog|->diagnostic|appendDiagnostic :: session-log lines a user is not shown
SKIP-LINE :: \bQUrl\b|\bQUrlQuery\b|addQueryItem|setQuery|https?:// :: a network address or query parameter
SKIP-LINE :: \bQRegularExpression\b :: a pattern is not wording
SKIP-LINE :: \bQSqlQuery\b|\bQSqlDatabase\b :: SQL statement text
SKIP-LINE :: \b(?:SELECT|INSERT\s+INTO|UPDATE\s+\w+\s+SET|CREATE\s+(?:TABLE|INDEX)|ALTER\s+TABLE|DROP\s+TABLE|PRAGMA)\b :: SQL statement text
SKIP-LINE :: \bsetObjectName\b|\bsetProperty\b|\bQMetaType :: Qt plumbing identifiers
SKIP-LINE :: \bqEnvironmentVariable|\bgetenv\b :: environment variable names
SKIP-LINE :: Q_TEST|\bQCOMPARE\b|\bQVERIFY\b :: test assertions

SKIP-LINE :: [\\]n" :: the tail of a printf format string that ends in a newline
